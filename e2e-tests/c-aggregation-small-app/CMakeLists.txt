cmake_minimum_required(VERSION 3.21)
project(profiling_small_app LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(profiling_small_app main.cpp)
target_include_directories(profiling_small_app PRIVATE "${PROFILER_SOURCE}/src/dd-win-prof/include")
option(SMALL_APP_NO_PROFILER "Build without loading the profiler DLL" OFF)
target_link_libraries(profiling_small_app PRIVATE psapi)
if(SMALL_APP_NO_PROFILER)
  target_compile_definitions(profiling_small_app PRIVATE SMALL_APP_NO_PROFILER)
else()
  target_link_libraries(profiling_small_app PRIVATE "${PROFILER_BINARY}/dd-win-prof.lib")
  add_custom_command(TARGET profiling_small_app POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PROFILER_BINARY}/dd-win-prof.dll" "$<TARGET_FILE_DIR:profiling_small_app>")
endif()
