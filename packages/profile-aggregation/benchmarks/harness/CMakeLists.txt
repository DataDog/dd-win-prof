cmake_minimum_required(VERSION 3.21)
project(aggregation_perf_v2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PA_BENCH_MSVC_RUNTIME "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING
    "Matching CRT for every benchmark target; verify baseline archive directives")
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "${PA_BENCH_MSVC_RUNTIME}")
endif()
get_filename_component(PA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(PA_BENCH_V19 "Build separate v19 CFFI executable (no fetching/building Rust)" OFF)
set(PA_V19_ROOT "" CACHE PATH "Existing libdatadog v19 binary package, containing include/datadog")
set(PA_V19_LIBRARY "" CACHE FILEPATH "Explicit existing v19 library/import library")
set(PA_V19_DLL "" CACHE FILEPATH "Existing v19 DLL, if using an import library")
option(PA_BENCH_V47 "Build separate v47 public CFFI executable" OFF)
set(PA_V47_ROOT "" CACHE PATH "Explicit libdatadog v47.0.0 release package root")
set(PA_V47_LIBRARY "" CACHE FILEPATH "Explicit existing v47 library/import library")
set(PA_V47_DLL "" CACHE FILEPATH "Existing v47 DLL, if using an import library")
if(PA_BENCH_V19 AND PA_BENCH_V47)
  message(FATAL_ERROR "Use separate build directories for v19 and v47 results")
endif()
if(PA_BENCH_V19)
  add_compile_definitions(PA_BENCH_BASELINE_VERSION=19)
else()
  add_compile_definitions(PA_BENCH_BASELINE_VERSION=47)
endif()
include("${PA_ROOT}/vendor/symbols.cmake")
foreach(variant IN ITEMS core core_timed)
  add_library(${variant}_bench_library STATIC "${PA_ROOT}/src/profile_aggregation.c"
    "${PA_ROOT}/vendor/pb_common.c" "${PA_ROOT}/vendor/pb_encode.c" "${PA_ROOT}/vendor/zstd.c")
  target_include_directories(${variant}_bench_library PUBLIC "${PA_ROOT}/include" PRIVATE "${PA_ROOT}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(${variant}_bench_library PRIVATE ${PA_VENDOR_SYMBOLS})
  add_executable(bench_${variant} driver.cpp)
  target_link_libraries(bench_${variant} PRIVATE ${variant}_bench_library)
  if(WIN32)
    target_link_libraries(bench_${variant} PRIVATE psapi)
  endif()
endforeach()
target_compile_definitions(core_timed_bench_library PRIVATE PA_BENCH_COMPRESSION_HOOKS)
target_compile_definitions(bench_core_timed PRIVATE PA_BENCH_COMPRESSION_HOOKS)
foreach(version IN ITEMS 19 47)
  if(PA_BENCH_V${version})
    if(NOT EXISTS "${PA_V${version}_ROOT}/include/datadog/profiling.h" OR NOT EXISTS "${PA_V${version}_LIBRARY}")
      message(FATAL_ERROR "Supply explicit PA_V${version}_ROOT and PA_V${version}_LIBRARY; no _deps discovery")
    endif()
    file(READ "${PA_V${version}_ROOT}/include/datadog/profiling.h" baseline_header)
    if(version EQUAL 47 AND NOT baseline_header MATCHES "ddog_prof_Profile_set_custom_sample_type")
      message(FATAL_ERROR "PA_V47_ROOT does not expose v47 public sample-type API")
    endif()
    add_executable(bench_v${version} driver.cpp)
    target_compile_definitions(bench_v${version} PRIVATE PA_BENCH_V${version})
    target_include_directories(bench_v${version} PRIVATE "${PA_V${version}_ROOT}/include")
    target_link_libraries(bench_v${version} PRIVATE "${PA_V${version}_LIBRARY}")
    if(WIN32)
      target_link_libraries(bench_v${version} PRIVATE psapi ntdll userenv bcrypt crypt32 wsock32 ws2_32 shlwapi secur32 ncrypt powrprof Version iphlpapi)
    endif()
    if(PA_V${version}_DLL)
      add_custom_command(TARGET bench_v${version} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PA_V${version}_DLL}" "$<TARGET_FILE_DIR:bench_v${version}>")
    endif()
  endif()
endforeach()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME result_version_invariants COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_results.py")
  add_test(NAME harness_invariants COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_harness.py" "$<TARGET_FILE:bench_core>" "$<TARGET_FILE:bench_core_timed>")
endif()
