cmake_minimum_required(VERSION 3.21)
project(aggregation_perf_v2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PA_BENCH_MSVC_RUNTIME "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING
    "Matching CRT for every benchmark target; v19 release/static uses /MT")
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "${PA_BENCH_MSVC_RUNTIME}")
endif()
get_filename_component(PA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(PA_BENCH_V19 "Build separate v19 CFFI executable (no fetching/building Rust)" OFF)
set(PA_V19_ROOT "" CACHE PATH "Existing libdatadog v19 binary package, containing include/datadog")
set(PA_V19_LIBRARY "" CACHE FILEPATH "Explicit existing v19 library/import library")
set(PA_V19_DLL "" CACHE FILEPATH "Existing v19 DLL, if using an import library")
include("${PA_ROOT}/vendor/symbols.cmake")
foreach(variant IN ITEMS core core_timed)
  add_library(${variant}_bench_library STATIC "${PA_ROOT}/src/profile_aggregation.c"
    "${PA_ROOT}/vendor/pb_common.c" "${PA_ROOT}/vendor/pb_encode.c" "${PA_ROOT}/vendor/zstd.c")
  target_include_directories(${variant}_bench_library PUBLIC "${PA_ROOT}/include" PRIVATE "${PA_ROOT}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(${variant}_bench_library PRIVATE ${PA_VENDOR_SYMBOLS})
  add_executable(bench_${variant} driver.cpp)
  target_link_libraries(bench_${variant} PRIVATE ${variant}_bench_library)
  if(WIN32)
    target_link_libraries(bench_${variant} PRIVATE psapi)
  endif()
endforeach()
target_compile_definitions(core_timed_bench_library PRIVATE PA_BENCH_COMPRESSION_HOOKS)
target_compile_definitions(bench_core_timed PRIVATE PA_BENCH_COMPRESSION_HOOKS)
if(PA_BENCH_V19)
  if(NOT EXISTS "${PA_V19_ROOT}/include/datadog/profiling.h" OR NOT EXISTS "${PA_V19_LIBRARY}")
    message(FATAL_ERROR "Supply PA_V19_ROOT and PA_V19_LIBRARY pointing to an existing v19 binary package")
  endif()
  add_executable(bench_v19 driver.cpp)
  target_compile_definitions(bench_v19 PRIVATE PA_BENCH_V19)
  target_include_directories(bench_v19 PRIVATE "${PA_V19_ROOT}/include")
  target_link_libraries(bench_v19 PRIVATE "${PA_V19_LIBRARY}")
  if(WIN32)
    target_link_libraries(bench_v19 PRIVATE psapi ntdll userenv bcrypt crypt32 wsock32 ws2_32 shlwapi secur32 ncrypt powrprof)
  endif()
  if(PA_V19_DLL)
    add_custom_command(TARGET bench_v19 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PA_V19_DLL}" "$<TARGET_FILE_DIR:bench_v19>")
  endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME harness_invariants COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_harness.py" "$<TARGET_FILE:bench_core>" "$<TARGET_FILE:bench_core_timed>")
endif()
