cmake_minimum_required(VERSION 3.21)
project(profile_aggregation LANGUAGES C)
option(PA_BUILD_TESTS "Build standalone aggregation tests" ${PROJECT_IS_TOP_LEVEL})
option(PA_BUILD_BENCHMARK "Build ingestion/serialization benchmark" ${PROJECT_IS_TOP_LEVEL})
add_library(profile_aggregation STATIC src/profile_aggregation.c vendor/pb_common.c vendor/pb_encode.c vendor/zstd.c)
target_include_directories(profile_aggregation PUBLIC include PRIVATE vendor)
target_compile_features(profile_aggregation PUBLIC c_std_11)
set_target_properties(profile_aggregation PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
# Private static dependencies must not interpose an embedding SDK's nanopb/zstd.
# Prefix every externally defined vendor symbol, including experimental zstd APIs.
include(vendor/symbols.cmake)
target_compile_definitions(profile_aggregation PRIVATE ${PA_VENDOR_SYMBOLS})
if(PA_BUILD_TESTS)
  enable_testing()
  # Independent decoder, test-only. Production contains only the pinned encoder.
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(PA_TEST_ZSTD QUIET IMPORTED_TARGET libzstd>=1.5.7)
  endif()
  if(TARGET PkgConfig::PA_TEST_ZSTD)
    set(PA_TEST_DECODER PkgConfig::PA_TEST_ZSTD)
  else()
    include(FetchContent)
    set(ZSTD_BUILD_PROGRAMS OFF)
    set(ZSTD_BUILD_TESTS OFF)
    set(ZSTD_BUILD_SHARED OFF)
    set(ZSTD_BUILD_STATIC ON)
    FetchContent_Declare(pa_test_zstd
      URL https://github.com/facebook/zstd/releases/download/v1.5.7/zstd-1.5.7.tar.gz
      URL_HASH SHA256=eb33e51f49a15e023950cd7825ca74a4a2b43db8354825ac24fc1b7ee09e6fa3
      SOURCE_SUBDIR build/cmake)
    FetchContent_MakeAvailable(pa_test_zstd)
    # Test dependency must never add install artifacts to the profiler package.
    set_property(DIRECTORY "${pa_test_zstd_SOURCE_DIR}/build/cmake" PROPERTY EXCLUDE_FROM_ALL TRUE)
    set(PA_TEST_DECODER libzstd_static)
  endif()
  add_executable(aggregation_tests tests/test.c)
  target_link_libraries(aggregation_tests PRIVATE profile_aggregation ${PA_TEST_DECODER})
  add_library(profile_aggregation_collisions STATIC src/profile_aggregation.c vendor/pb_common.c vendor/pb_encode.c vendor/zstd.c)
  target_include_directories(profile_aggregation_collisions PUBLIC include PRIVATE vendor)
  target_compile_features(profile_aggregation_collisions PUBLIC c_std_11)
  target_compile_definitions(profile_aggregation_collisions PRIVATE PA_TEST_COLLISIONS ${PA_VENDOR_SYMBOLS})
  add_executable(aggregation_collision_tests tests/test.c)
  target_compile_definitions(aggregation_collision_tests PRIVATE PA_TEST_COLLISIONS)
  target_link_libraries(aggregation_collision_tests PRIVATE profile_aggregation_collisions ${PA_TEST_DECODER})
  add_test(NAME aggregation_collision_tests COMMAND aggregation_collision_tests)
  add_test(NAME aggregation_tests COMMAND aggregation_tests)
endif()
if(PA_BUILD_BENCHMARK)
  add_executable(aggregation_bench tests/bench.c)
  target_link_libraries(aggregation_bench PRIVATE profile_aggregation)
endif()
if(PA_BUILD_BENCHMARK AND WIN32)
  target_link_libraries(aggregation_bench PRIVATE psapi)
endif()
