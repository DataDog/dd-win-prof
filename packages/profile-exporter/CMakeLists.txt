cmake_minimum_required(VERSION 3.21)
project(profile_exporter LANGUAGES C)
option(PE_BUILD_TESTS "Build standalone exporter tests" ${PROJECT_IS_TOP_LEVEL})
add_library(profile_exporter STATIC src/protocol.c)
target_include_directories(profile_exporter PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_compile_features(profile_exporter PUBLIC c_std_11)
set_target_properties(profile_exporter PROPERTIES C_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_sources(profile_exporter PRIVATE src/winhttp.c)
  target_link_libraries(profile_exporter PUBLIC winhttp)
  if(MSVC)
    target_link_libraries(profile_exporter PUBLIC delayimp)
    target_link_options(profile_exporter INTERFACE /DELAYLOAD:winhttp.dll)
  endif()
endif()
if(PE_BUILD_TESTS)
  enable_testing()
  add_executable(exporter_protocol_tests tests/protocol_test.c)
  target_include_directories(exporter_protocol_tests PRIVATE include)
  target_compile_features(exporter_protocol_tests PRIVATE c_std_11)
  add_test(NAME exporter_protocol_tests COMMAND exporter_protocol_tests)
  if(WIN32)
    add_executable(exporter_probe tests/probe.c)
    target_link_libraries(exporter_probe PRIVATE profile_exporter)
  endif()
endif()
